using System;
using System.IO;

class Program
{
    static void Main()
    {
        // Przekierowanie Console.Out do pliku HTML
        using (StreamWriter writer = new StreamWriter("output.html"))
        {
            Console.SetOut(writer);
            Console.WriteLine("<html>");
            Console.WriteLine("<head><title>Console Output</title></head>");
            Console.WriteLine("<body>");
            Console.WriteLine("<h1>Wynik z konsoli</h1>");
            Console.WriteLine("<p>To jest przekierowany tekst z Console.WriteLine()</p>");
            Console.WriteLine("</body>");
            Console.WriteLine("</html>");
        }

        Console.WriteLine("Plik HTML został wygenerowany!");
    }
}


--------------

using System.IO;
using System.Web.Mvc;

public class HomeController : Controller
{
    public ActionResult Output()
    {
        string filePath = Server.MapPath("~/output.html");
        string content = System.IO.File.ReadAllText(filePath);
        return Content(content, "text/html");
    }
}
